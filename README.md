  function addToCartAndGo(name, price) {
    let cart = JSON.parse(localStorage.getItem("cart")) || [];

    cart.push({
      name: name,
      price: price,
      qty: 1
    });

    localStorage.setItem("cart", JSON.stringify(cart));

    // 👉 พาไปหน้าตะกร้า
    window.location.href = "7.html";
  }
